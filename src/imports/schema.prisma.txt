generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")
}

enum Role {
  MUSICIAN
  CONTRACTOR
}

enum EventStatus {
  ABERTO
  CONTRATADO
  CANCELADO
  CONCLUIDO
}

model User {
  id                String             @id @default(uuid())
  email             String             @unique
  passwordHash      String
  role              Role
  uf                String
  cidade            String
  createdAt         DateTime           @default(now())
  updatedAt         DateTime           @updatedAt
  musicianProfile   MusicianProfile?
  contractorProfile ContractorProfile?
}

model MusicianProfile {
  id              String   @id @default(uuid())
  userId          String   @unique
  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  nomeCompleto    String
  cpf             String   @unique
  telefone        String
  fotoPerfilUrl   String
  bio             String?
  estilosMusicais String[]
  uf              String
  cidade          String
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  hiredEvents Event[] @relation("MusicoContratado")
}

model ContractorProfile {
  id            String   @id @default(uuid())
  userId        String   @unique
  user          User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  nomeFantasia  String
  cnpj          String   @unique
  telefone      String
  fotoPerfilUrl String
  uf            String
  cidade        String
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  events Event[]
}

model BlockedDocument {
  id        String   @id @default(uuid())
  documento String   @unique
  tipo      Role
  blockedAt DateTime @default(now())
}

model Event {
  id                 String            @id @default(uuid())
  contractorId       String
  contractor         ContractorProfile @relation(fields: [contractorId], references: [id], onDelete: Cascade)
  
  musicoContratadoId String?
  musicoContratado   MusicianProfile?  @relation("MusicoContratado", fields: [musicoContratadoId], references: [id], onDelete: SetNull)

  titulo        String
  tipoEvento    String
  data          DateTime
  horarioInicio String
  duracao       String
  uf            String
  cidade        String
  estiloMusical String
  cache         Float
  imagemUrl     String
  descricao     String?
  status        EventStatus @default(ABERTO)
  
  createdAt     DateTime    @default(now())
  updatedAt     DateTime    @updatedAt
}
